/* ==========================================================
 * PARAMETER BUILDER
 * ========================================================== */


function buildDlpdParams(
    customerType: CustomerType,
    month?: string,
    filters?: DlpdQueryFilters,
) {
    return {
        customer_type:
            customerType,

        ...(month
            ? {
                  month,
              }
            : {}),

        ...(filters?.unitupi
            ? {
                  unitupi:
                      filters.unitupi,
              }
            : {}),

        ...(filters?.unitap
            ? {
                  unitap:
                      filters.unitap,
              }
            : {}),

        ...(filters?.unitup
            ? {
                  unitup:
                      filters.unitup,
              }
            : {}),

        ...(filters?.tariff
            ? {
                  tariff:
                      filters.tariff,
              }
            : {}),

        /*
         * STATUS HASIL
         *
         * NORMAL / TEMUAN
         */
        ...(filters?.status
            ? {
                  status:
                      filters.status,
              }
            : {}),

        /*
         * STATUS PEMERIKSAAN
         *
         * SUDAH PERIKSA / BELUM PERIKSA
         */
        ...(filters?.inspection_status
            ? {
                  inspection_status:
                      filters.inspection_status,
              }
            : {}),

        ...(filters?.dlpd_repeat
            ? {
                  dlpd_repeat:
                      filters.dlpd_repeat,
              }
            : {}),

        /*
         * KENDALA
         *
         * Dikirim sebagai parameter:
         *
         * kendala=<value>
         *
         * Prabayar:
         * kendala=<nilai KETERANGAN>
         *
         * Pascabayar:
         * kendala=KWH < 40 JAM
         *
         * atau:
         * kendala=KWH NOL
         */
        ...(filters?.kendala
            ? {
                  kendala:
                      filters.kendala,
              }
            : {}),

        ...(filters?.search_idpel
            ? {
                  search_idpel:
                      filters.search_idpel,
              }
            : {}),

        ...(filters?.search_nama
            ? {
                  search_nama:
                      filters.search_nama,
              }
            : {}),
    };
}

function isValidMonthKey(
    value: unknown,
): value is string {
    if (
        typeof value !== "string"
    ) {
        return false;
    }

    return /^\d{6}$/.test(
        value,
    );
}


function normalizeMonths(
    result:
        | MonthOption[]
        | undefined
        | null,
): MonthOption[] {
    if (
        !Array.isArray(result)
    ) {
        return [];
    }

    const unique =
        new Map<
            string,
            MonthOption
        >();

    for (
        const item of result
    ) {
        if (
            !item ||
            !isValidMonthKey(
                item.month_key,
            )
        ) {
            continue;
        }

        unique.set(
            item.month_key,
            {
                month_key:
                    item.month_key,
                label:
                    item.label ||
                    item.month_key,
            },
        );
    }

    return Array.from(
        unique.values(),
    ).sort(
        (a, b) =>
            a.month_key.localeCompare(
                b.month_key,
            ),
    );
}


function normalizeOptions(
    values:
        | string[]
        | undefined
        | null,
): string[] {
    if (
        !Array.isArray(values)
    ) {
        return [];
    }

    return Array.from(
        new Set(
            values
                .filter(
                    (
                        value,
                    ) =>
                        typeof value ===
                            "string" &&
                        value.trim() !== "",
                )
                .map(
                    (
                        value,
                    ) =>
                        value.trim(),
                ),
        ),
    );
}

function formatValue(
    value: unknown,
): string {

    if (
        value === null ||
        value === undefined ||
        String(value).trim() === ""
    ) {
        return "-";
    }

    return String(value);
}


function formatNumber(
    value: unknown,
): string {

    if (
        value === null ||
        value === undefined ||
        String(value).trim() === ""
    ) {
        return "-";
    }


    const numeric =
        Number(value);


    if (
        !Number.isFinite(
            numeric,
        )
    ) {
        return String(value);
    }


    return numeric.toLocaleString(
        "id-ID",
    );
}


function statusClass(
    status: string | null,
): string {

    const normalized =
        String(
            status ?? "",
        )
            .trim()
            .toLowerCase();


    if (
        normalized === "normal"
    ) {
        return "status-badge status-normal";
    }


    if (
        normalized === "temuan"
    ) {
        return "status-badge status-temuan";
    }


    return "status-badge status-belum";
}

import "./DlpdDetail.css";

import {
    useEffect,
    useState,
} from "react";

import {
    getDlpdCustomerDetail,
} from "../../api/dlpd";

import type {
    CustomerType,
    DlpdCustomerDetail,
} from "../../api/dlpd";

interface Props {
    idpel?: string;
    customerType: CustomerType;
    month?: string;
}

export default function DlpdDetail({
    idpel,
    customerType,
    month,
}: Props) {
    const [
        data,
        setData,
    ] = useState<
        DlpdCustomerDetail | null
    >(null);

    const [
        loading,
        setLoading,
    ] = useState<boolean>(false);

    useEffect(() => {
        if (!idpel) {
            setData(null);
            setLoading(false);
            return;
        }

        let mounted = true;

        const load = async () => {
            try {
                setLoading(true);
                setData(null);

                const result =
                    await getDlpdCustomerDetail(
                        idpel,
                        customerType,
                        month,
                    );

                if (mounted) {
                    setData(result);
                }
            } catch (err) {
                console.error(
                    "Failed to load DLPD customer detail:",
                    err,
                );

                if (mounted) {
                    setData(null);
                }
            } finally {
                if (mounted) {
                    setLoading(false);
                }
            }
        };

        load();

        return () => {
            mounted = false;
        };
    }, [
        idpel,
        customerType,
        month,
    ]);

    if (!idpel) {
        return (
            <div className="detail-empty">
                <h3>
                    Belum ada pelanggan dipilih
                </h3>

                <p>
                    Pilih pelanggan pada tabel
                    untuk melihat detail.
                </p>
            </div>
        );
    }

    if (loading) {
        return (
            <div className="detail-empty">
                <h3>
                    Loading...
                </h3>
            </div>
        );
    }

    if (!data?.customer) {
        return (
            <div className="detail-empty">
                <h3>
                    Data tidak ditemukan
                </h3>

                <p>
                    Detail pelanggan tidak
                    tersedia untuk data yang
                    dipilih.
                </p>
            </div>
        );
    }

    const customer =
        data.customer;

    return (
        <div className="detail-grid">

            <div>
                <span>IDPEL</span>

                <strong>
                    {customer.idpel ?? "-"}
                </strong>
            </div>

            <div>
                <span>Nama</span>

                <strong>
                    {customer.nama ?? "-"}
                </strong>
            </div>

            <div>
                <span>UNITUP</span>

                <strong>
                    {customer.unitup ?? "-"}
                </strong>
            </div>

            <div>
                <span>Status</span>

                <strong>
                    {customer.status ?? "-"}
                </strong>
            </div>

            <div>
                <span>Tarif</span>

                <strong>
                    {customer.tariff ?? "-"}
                </strong>
            </div>

            <div>
                <span>Daya</span>

                <strong>
                    {customer.daya != null
                        ? customer.daya.toLocaleString()
                        : "-"}
                </strong>
            </div>

            <div>

    DlpdMapPoint,

interface DlpdMapProps {
    points: DlpdMapPoint[];
    height?: number;
}


/* ==========================================================
 * DEFAULT CENTER
 * ========================================================== */

const DEFAULT_CENTER: [number, number] = [
    -5.3971,
    105.2668,
];


/* ==========================================================
 * MAP FITTER
 * ========================================================== */

function MapFitter({
    points,
}: {
    points: DlpdMapPoint[];
}) {
    const map = useMap();

    const bounds = useMemo(
        () => {
            if (points.length === 0) {
                return null;
            }

            return points.map(
                (point) =>
                    [
                        point.latitude,
                        point.longitude,
                    ] as [number, number],
            );
        },
        [points],
    );

    useEffect(
        () => {
            if (!bounds || bounds.length === 0) {
                map.setView(
                    DEFAULT_CENTER,
                    11,
                );

                return;
            }

            map.fitBounds(
                bounds,
                {
                    padding: [40, 40],
                    maxZoom: 15,
                },
            );
        },
        [map, bounds],
    );

    return null;
}


/* ==========================================================
 * STATUS COLOR
 * ========================================================== */

function getPointColor(
    point: DlpdMapPoint,
): string {
    const status =
        String(
            point.status ?? "",
        ).toUpperCase();

    if (
        status.includes("TEMUAN")
    ) {
        return "#ef4444";
    }

    if (
        status.includes("NORMAL")
    ) {
        return "#22c55e";
    }

    return "#3b82f6";
}

 * COMPONENT
 * ========================================================== */

                }}
            >
                <TileLayer
                    attribution='&copy; OpenStreetMap contributors'
                    url="https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png"
                />

                <MapFitter
                    points={points}
                />

                {points.map(
                    (
                        point,

                            center={[
